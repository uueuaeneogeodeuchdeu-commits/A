local connections = {}
local tickConnection = nil
local destroyed = false
local running = false
local collapsed = false
local accumulated = 0
local startedAt = 0
local lastLapTotal = 0
local lapCount = 0
local bestLap = math.huge
local bestLapIndex = nil
local lapRows = {}

local function listen(signal, callback)
    local conn = signal:Connect(callback)
    table.insert(connections, conn)
    return conn
end

local function elapsed()
    if running then return accumulated + math.max(0, os.clock() - startedAt) end
    return accumulated
end

local function formatTime(seconds)
    local ticks = math.floor(math.max(0, seconds) * 100 + 0.00001)
    local hundredths = ticks % 100
    local totalSeconds = math.floor(ticks / 100)
    local h = math.floor(totalSeconds / 3600)
    local m = math.floor(totalSeconds / 60) % 60
    local s = totalSeconds % 60
    return string.format("%02d:%02d:%02d.%02d", h, m, s, hundredths)
end

local function formatSplit(seconds)
    local ticks = math.floor(math.max(0, seconds) * 100 + 0.00001)
    local cs = ticks % 100
    local whole = math.floor(ticks / 100)
    return string.format("%02d:%02d.%02d", math.floor(whole / 60), whole % 60, cs)
end

local function refreshTime()
    local value = formatTime(elapsed())
    if timer.Text ~= value then timer.Text = value end
end

local function refreshButtons()
    status.Text = running and "●  RUNNING" or (accumulated > 0 and "●  PAUSED" or "●  READY")
    status.TextColor3 = running and C.green or (accumulated > 0 and C.amber or C.purple)
    startButton.Text = running and "Ⅱ  PAUSE" or (accumulated > 0 and "▶  RESUME" or "▶  START")
    lapButton.TextColor3 = running and C.white or C.muted
    lapButton.BackgroundColor3 = running and C.purpleSoft or C.card
    fine.Text = running and "MEASURING  ·  1/100s RESOLUTION" or "1/100s  ·  HIGH-PRECISION CLOCK"
    refreshTime()
end

local function stopTicker()
    if tickConnection then tickConnection:Disconnect(); tickConnection = nil end
end

local function startTicker()
    stopTicker()
    local previous = 0
    tickConnection = RunService.Heartbeat:Connect(function()
        local now = os.clock()
        if now - previous >= (1 / 30) then
            previous = now
            refreshTime()
        end
    end)
end

local function addLap(total, split)
    lapCount = lapCount + 1
    if split < bestLap then
        bestLap = split
        bestLapIndex = lapCount
    end
    lapCountLabel.Text = string.format("%d LAPS  ·  BEST %s", lapCount, formatSplit(bestLap))
    empty.Visible = false

    local row = new("Frame", {
        Name = "Lap" .. lapCount, LayoutOrder = -lapCount,
        Size = UDim2.new(1, -3, 0, 33),
        BackgroundColor3 = C.row, BackgroundTransparency = 0.28,
        BorderSizePixel = 0,
    }, scrolling)
    round(row, 7)
    local numberLabel = textLabel(row, string.format("#%02d", lapCount), 7, 0, 53, 33, 11, C.muted, true)
    local splitLabel = textLabel(row, formatSplit(split), 63, 0, 115, 33, 12, C.white, true)
    textLabel(row, formatTime(total), 180, 0, 125, 33, 11, C.muted, false, Enum.TextXAlignment.Right)
    lapRows[lapCount] = {index = numberLabel, split = splitLabel}

    for index, labels in pairs(lapRows) do
        local isBest = index == bestLapIndex
        labels.index.TextColor3 = isBest and C.green or C.muted
        labels.split.TextColor3 = isBest and C.green or C.white
    end
    scrolling.CanvasPosition = Vector2.new(0, 0)
end

listen(startButton.MouseButton1Click, function()
    if running then
        accumulated = elapsed()
        running = false
        stopTicker()
    else
        startedAt = os.clock()
        running = true
        startTicker()
    end
    refreshButtons()
end)

listen(lapButton.MouseButton1Click, function()
    if not running or lapCount >= 150 then return end
    local total = elapsed()
    local split = total - lastLapTotal
    lastLapTotal = total
    addLap(total, split)
end)

listen(resetButton.MouseButton1Click, function()
    running = false
    stopTicker()
    accumulated = 0
    startedAt = 0
    lastLapTotal = 0
    lapCount = 0
    bestLap = math.huge
    bestLapIndex = nil
    for _, row in pairs(lapRows) do
        if row.index.Parent then row.index.Parent:Destroy() end
    end
    table.clear(lapRows)
    lapCountLabel.Text = "0 LAPS"
    scrolling.CanvasPosition = Vector2.new(0, 0)
    empty.Visible = true
    refreshButtons()
end)

listen(collapseButton.MouseButton1Click, function()
    collapsed = not collapsed
    collapseButton.Text = collapsed and "+" or "−"
    lapsBlock.Visible = not collapsed
    footer.Visible = not collapsed
    TweenService:Create(main, TweenInfo.new(0.22, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {

end)

local function cleanup()
    if destroyed then return end
    destroyed = true
    stopTicker()
    for _, conn in ipairs(connections) do conn:Disconnect() end
    if gui.Parent then gui:Destroy() end
    if environment.__CHRONO_STOPWATCH_CLEANUP == cleanup then
        environment.__CHRONO_STOPWATCH_CLEANUP = nil
    end
end
environment.__CHRONO_STOPWATCH_CLEANUP = cleanup
listen(closeButton.MouseButton1Click, cleanup)

-- Dragging works with mouse and touchscreen; controls remain independently clickable.
local dragging = false
local dragStart = nil
local initialCenter = nil
local dragInput = nil
listen(header.InputBegan, function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragInput = input
        dragStart = input.Position
        initialCenter = main.AbsolutePosition + main.AbsoluteSize / 2
    end
end)
listen(UserInputService.InputChanged, function(input)
    if not dragging or not dragStart then return end
    local isTouch = dragInput.UserInputType == Enum.UserInputType.Touch and input == dragInput
    local isMouse = dragInput.UserInputType == Enum.UserInputType.MouseButton1
        and input.UserInputType == Enum.UserInputType.MouseMovement
    if not (isTouch or isMouse) then return end

    local delta = input.Position - dragStart
    local center = initialCenter + Vector2.new(delta.X, delta.Y)
    local camera = workspace.CurrentCamera
    if camera then
        local view = camera.ViewportSize
        local half = main.AbsoluteSize / 2
        center = Vector2.new(
            math.clamp(center.X, half.X, math.max(half.X, view.X - half.X)),
            math.clamp(center.Y, half.Y, math.max(half.Y, view.Y - half.Y))
        )
    end
    main.Position = UDim2.fromOffset(center.X, center.Y)
end)
listen(UserInputService.InputEnded, function(input)
    if input == dragInput then
        dragging = false
        dragInput = nil
        dragStart = nil
    end
end)

refreshButtons()
